create or replace function public.odeme_bildir( p_paket text, p_sure_gun integer, p_ilan_hakki integer, p_tutar numeric, p_gonderen_ad_soyad text, p_dekont_no text default null, p_aciklama text default null ) returns bigint language plpgsql security definer set search_path = public as $$ declare v_user uuid := auth.uid(); v_baslangic timestamptz; v_bitis timestamptz; begin if not exists ( select 1 from public.admin_kullanicilar where user_id = auth.uid() ) then raise exception 'Yönetici yetkisi gerekli'; end if; select * into v_odeme from public.odeme_talepleri where id = p_odeme_id for update; update public.odeme_talepleri set durum = 'onaylandi', onaylayan = auth.uid(), onay_tarihi = now() where id = p_odeme_id; return true; end; $$; revoke all on function public.odeme_onayla(bigint) from public;
